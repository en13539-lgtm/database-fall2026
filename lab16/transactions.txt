Transaction best practices
Keep transactions short and focused
lab16_transactions=# BEGIN;

UPDATE inventory
SET stock = stock - 1
WHERE product_id = 101;

INSERT INTO order_items (order_id, product_id, quantity)
VALUES (1, 101, 1);

COMMIT;
BEGIN
UPDATE 1
INSERT 0 1
COMMIT


Handle errors properly
lab16_transactions=# BEGIN;

DO $$
DECLARE
    current_balance DECIMAL;
BEGIN
    SELECT balance INTO current_balance
    FROM accounts
    WHERE account_id = 1;

    IF current_balance < 100 THEN
        RAISE EXCEPTION 'Insufficient funds';
    END IF;

    UPDATE accounts
    SET balance = balance - 100
    WHERE account_id = 1;

    UPDATE accounts
    SET balance = balance + 100
    WHERE account_id = 2;

EXCEPTION
    WHEN OTHERS THEN
        RAISE NOTICE 'Transaction failed: %', SQLERRM;
END $$;

COMMIT;
BEGIN
DO
COMMIT


Use appropriate isolation levels
1. SERIALIZABLE
lab16_transactions=# BEGIN TRANSACTION ISOLATION LEVEL SERIALIZABLE;

SELECT * FROM accounts;

COMMIT;
BEGIN
 account_id | account_holder | balance 
------------+----------------+---------
          3 | Charlie        | 1575.00
          4 | Diana          |  840.00
          1 | Alice          | 4420.00
          2 | Bob            | 3980.00
(4 rows)

COMMIT

2. READ COMMITTED
lab16_transactions=# BEGIN TRANSACTION ISOLATION LEVEL READ COMMITTED;

SELECT * FROM accounts;

COMMIT;
BEGIN
 account_id | account_holder | balance 
------------+----------------+---------
          3 | Charlie        | 1575.00
          4 | Diana          |  840.00
          1 | Alice          | 4420.00
          2 | Bob            | 3980.00
(4 rows)

COMMIT



Use savepoints for complex operations
lab16_transactions=# BEGIN;

INSERT INTO orders (customer_id, total)
VALUES (1, 1000.00);

SAVEPOINT before_items;

INSERT INTO order_items (order_id, product_id, quantity)
VALUES (currval('orders_order_id_seq'), 101, 2);

INSERT INTO order_items (order_id, product_id, quantity)
VALUES (currval('orders_order_id_seq'), 102, 1);

COMMIT;
BEGIN
INSERT 0 1
SAVEPOINT
INSERT 0 1
INSERT 0 1
COMMIT


Monitor transaction locks and deadlocks
lab16_transactions=# SELECT
    blocked_locks.pid AS blocked_pid,
    blocked_activity.usename AS blocked_user,
    blocking_locks.pid AS blocking_pid,
    blocking_activity.usename AS blocking_user,
    blocked_activity.query AS blocked_statement
FROM pg_catalog.pg_locks blocked_locks
JOIN pg_catalog.pg_stat_activity blocked_activity
    ON blocked_activity.pid = blocked_locks.pid
JOIN pg_catalog.pg_locks blocking_locks
    ON blocking_locks.locktype = blocked_locks.locktype
    AND blocking_locks.pid <> blocked_locks.pid
    AND blocking_locks.granted
JOIN pg_catalog.pg_stat_activity blocking_activity
    ON blocking_activity.pid = blocking_locks.pid
WHERE NOT blocked_locks.granted;
 blocked_pid | blocked_user | blocking_pid | blocking_user | blocked_statement 
-------------+--------------+--------------+---------------+-------------------
(0 rows)
