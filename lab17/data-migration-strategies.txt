Strategy 4: ETL Pipeline Migration
lab17_backup=# CREATE TABLE staging_employees AS
SELECT * FROM employees WHERE 1=0;
SELECT 0
lab17_backup=# \copy (SELECT * FROM employees LIMIT 10000 OFFSET 0) TO 'employees_batch_1.csv' WITH CSV HEADER
COPY 5
lab17_backup=# \copy staging_employees FROM 'employees_batch_1.csv' WITH CSV HEADER
COPY 5
lab17_backup=# SELECT COUNT(*) FROM staging_employees;

lab17_backup=# INSERT INTO employees
SELECT * FROM staging_employees

    first_name = EXCLUDED.first_name,
